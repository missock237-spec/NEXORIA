    28→// ── RNG déterministe ──
    29→function mulberry32(seed: number) {
    30→  let a = seed >>> 0
    31→  return () => {
    32→    a |= 0
    33→    a = (a + 0x6d2b79f5) | 0
    34→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    35→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    36→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    37→  }
    38→}

   316→function BossFight({ encounter, onEnd }: { encounter: BossEncounter; onEnd: (r: 'victory' | 'defeat') => void }) {
   317→  const activeWorld = useCreatorStore((s) => s.activeWorld)
   318→  const playerRace = (
   319→    useCreatorStore((s) => s.worldConfig)?.races.find((r) => r.id === activeWorld?.character.race) ?? null
   320→  ) as RaceDef | null
   321→  const playerApp = activeWorld?.character.appearance ?? null
   322→  const playerEq = useMemo(() => {
   323→    const cfg = useCreatorStore.getState().worldConfig
   324→    const m: Record<string, import('@/lib/game/types').EquipmentDef> = {}
   325→    if (activeWorld && cfg) {
   326→      for (const [, id] of Object.entries(activeWorld.character.equipment)) {
   327→        const item = cfg.equipment.find((e) => e.id === id)
   328→        if (item) m[item.slot] = item as import('@/lib/game/types').EquipmentDef
   329→      }
   330→    }
   331→    return m
   332→  }, [activeWorld])
   333→
   334→  const stats = useRef({
   335→    maxHp: activeWorld?.character.stats.maxHp ?? 100,
   336→    atk: activeWorld?.character.stats.attaquePhysique ?? 10,
   337→  })
   338→
   339→  const playerG = useRef<THREE.Group>(null)
   340→  const cam = useRef({ yaw: 0, pitch: 0.52, dist: 8.6 })
   341→  const pos = useRef(new THREE.Vector3(0, 0, 7))
   342→  const moveRef = useRef(0)
   343→  const { camera, gl } = useThree()
   344→  const rand = useRef(mulberry32(encounter.seed))
   345→
   346→  const strikes = useRef<Strike[]>([])
   347→  const shockwaves = useRef<{ r: number; speed: number; hit: boolean }[]>([])
   348→  const [strikesView, setStrikesView] = useState<Strike[]>([])
   349→  const waveRefs = useRef<(THREE.Mesh | null)[]>([])
   350→  const [, force] = useState(0)
   351→
   352→  useEffect(() => {
   353→    bossCombat.bossHp = encounter.bossHp
   354→    bossCombat.bossMaxHp = encounter.bossHp
   355→    bossCombat.playerHp = stats.current.maxHp
   356→    bossCombat.playerMaxHp = stats.current.maxHp
   357→    bossCombat.running = true
   358→    bossCombat.over = false
   359→    bossCombat.result = null
   360→    bossCombat.phase = 1
   361→    bossCombat.phaseStart = 0
   362→    bossCombat.phaseTimes = []
   363→    bossCombat.window = false
   364→    bossCombat.telegraph = null
   365→    bossCombat.shockwave = 0
   366→    return () => {
   367→      bossCombat.running = false
   368→    }
   369→  }, [encounter])
   370→
   371→  useEffect(() => {
   372→    const el = gl.domElement
   373→    let dragging = false
   374→    let lx = 0
   375→    let ly = 0
   376→    const down = (e: PointerEvent) => {
   377→      dragging = true
   378→      lx = e.clientX
   379→      ly = e.clientY
   380→    }
   381→    const move = (e: PointerEvent) => {
   382→      if (!dragging) return
   383→      cam.current.yaw -= (e.clientX - lx) * 0.0052
   384→      cam.current.pitch = Math.max(0.06, Math.min(1.1, cam.current.pitch + (e.clientY - ly) * 0.0036))
   385→      lx = e.clientX
   386→      ly = e.clientY
   387→    }
   388→    const up = () => (dragging = false)
   389→    const click = (e: MouseEvent) => {
   390→      if (Math.abs(e.clientX - lx) < 4 && Math.abs(e.clientY - ly) < 4) requestAttack()
   391→    }
   392→    el.addEventListener('pointerdown', down)
   393→    window.addEventListener('pointermove', move)
   394→    window.addEventListener('pointerup', up)
   395→    el.addEventListener('click', click)
   396→    return () => {
   397→      el.removeEventListener('pointerdown', down)
   398→      window.removeEventListener('pointermove', move)
   399→      window.removeEventListener('pointerup', up)
   400→      el.removeEventListener('click', click)
   401→    }
   402→  }, [gl])
   403→
   404→  const hurtPlayer = useCallback((pct: number) => {
   405→    const d = Math.max(4, Math.round(stats.current.maxHp * pct))
   406→    bossCombat.playerHp = Math.max(0, bossCombat.playerHp - d)
   407→    bossCombat.hitFlash = Date.now()
   408→    if (bossCombat.playerHp <= 0 && !bossCombat.over) {
   409→      bossCombat.over = true
   410→      bossCombat.result = 'defeat'
   411→      onEnd('defeat')
   412→    }
   413→  }, [onEnd])
   414→
   415→  const hitBoss = useCallback(() => {
   416→    // Dégâts plafonnés : chaque fenêtre ne peut pas trancher une phase entière
   417→    const cap = bossCombat.bossMaxHp * 0.16
   418→    const d = Math.round(Math.min(cap, Math.max(8, stats.current.atk * 1.5)) * (0.9 + rand.current() * 0.2))
   419→    bossCombat.bossHp = Math.max(0, bossCombat.bossHp - d)
   420→    bossCombat.bossHitFlash = Date.now()
   421→    if (bossCombat.bossHp <= 0 && !bossCombat.over) {
   422→      bossCombat.over = true
   423→      bossCombat.result = 'victory'
   424→      bossCombat.window = false
   425→      onEnd('victory')
   426→    }
   427→  }, [onEnd])
   428→
   429→  const timers = useRef({ strike: 3.4, wave: 8, window: 2.2, windowDur: 0, force: 0 })
   430→
   431→  useFrame((_, rawDelta) => {
   432→    const dt = Math.min(rawDelta, 0.05)
   433→    if (!bossCombat.running || bossCombat.over) {
   434→      if (playerG.current) {
   435→        camera.position.lerp(new THREE.Vector3(pos.current.x, 4.5, pos.current.z + 11), 0.03)
   436→        camera.lookAt(0, 8, -6)
   437→      }
   438→      return
   439→    }
   440→    bossCombat.elapsed += dt
   441→    const T = timers.current
   442→
   443→    // ── Joueur ──
   444→    const mag = Math.hypot(playerInput.x, playerInput.z)
   445→    const yaw = cam.current.yaw
   446→    const speed = playerInput.sprint ? 8.6 : 5.8
   447→    if (mag > 0.08) {
   448→      const fx = Math.sin(yaw)
   449→      const fz = Math.cos(yaw)
   450→      const ax = -fz * playerInput.x + fx * -playerInput.z
   451→      const az = fx * playerInput.x + fz * -playerInput.z
   452→      const l = Math.hypot(ax, az) || 1
   453→      pos.current.x += (ax / l) * speed * dt
   454→      pos.current.z += (az / l) * speed * dt
   455→      moveRef.current = Math.min(1, moveRef.current + dt * 6)
   456→    } else {
   457→      moveRef.current = Math.max(0, moveRef.current - dt * 8)
   458→    }
   459→    // Gravité fluctuante (phase 2+) : aspiration vers le centre
   460→    if (bossCombat.phase >= 2) {
   461→      bossCombat.gravityPull = 0.22
   462→      const d = Math.hypot(pos.current.x, pos.current.z)
   463→      if (d > 2.4) {
   464→        pos.current.x -= (pos.current.x / d) * 2.1 * dt
   465→        pos.current.z -= (pos.current.z / d) * 2.1 * dt
   466→      }
   467→    } else {
   468→      bossCombat.gravityPull = 0
   469→    }
   470→    const r = Math.hypot(pos.current.x, pos.current.z)
   471→    if (r > PLATFORM_R - 0.9) {
   472→      pos.current.x = (pos.current.x / r) * (PLATFORM_R - 0.9)
   473→      pos.current.z = (pos.current.z / r) * (PLATFORM_R - 0.9)
   474→    }
   475→
   476→    // ── Fenêtres de vulnérabilité (le noyau s'expose) ──
   477→    T.window -= dt
   478→    if (T.window <= 0) {
   479→      bossCombat.window = true
   480→      T.windowDur = bossCombat.phase >= 3 ? 2.2 : 2.6
   481→      T.window = (bossCombat.phase >= 3 ? 4.4 : 5.4) + T.windowDur
   482→      setTimeout(() => (bossCombat.window = false), T.windowDur * 1000)
   483→    }
   484→
   485→    // ── Attaques du boss ──
   486→    const phase = bossCombat.phase
   487→    // Phase 1+ : éclairs ciblés
   488→    T.strike -= dt
   489→    if (T.strike <= 0) {
   490→      const interval = phase >= 3 ? 2.3 : phase >= 2 ? 3 : 3.8
   491→      T.strike = interval
   492→      const count = phase >= 3 ? 2 : 1
   493→      for (let i = 0; i < count; i++) {
   494→        const lead = 0.55
   495→        const x = THREE.MathUtils.clamp(
   496→          pos.current.x + playerInput.x * speed * lead * 0.5 + (rand.current() - 0.5) * 2.4,
   497→          -PLATFORM_R + 1.5, PLATFORM_R - 1.5
   498→        )
   499→        const z = THREE.MathUtils.clamp(
   500→          pos.current.z + playerInput.z * speed * lead * 0.5 + (rand.current() - 0.5) * 2.4,
   501→          -PLATFORM_R + 1.5, PLATFORM_R - 1.5
   502→        )
   503→        const warn = 1.25
   504→        strikes.current.push({ id: Date.now() + i, x, z, t: bossCombat.elapsed + warn, r: 1.6 })
   505→      }
   506→      setStrikesView([...strikes.current])
   507→    }
   508→    strikes.current = strikes.current.filter((s) => {
   509→      if (bossCombat.elapsed >= s.t) {
   510→        // Impact
   511→        if (Math.hypot(pos.current.x - s.x, pos.current.z - s.z) < s.r) hurtPlayer(0.13)
   512→        setStrikesView([...strikes.current.filter((x) => x.id !== s.id)])
   513→        return false
   514→      }
   515→      return true
   516→    })
   517→
   518→    // Phase 2+ : ondes de choc annulaires
   519→    if (phase >= 2) {
   520→      T.wave -= dt
   521→      if (T.wave <= 0) {
   522→        T.wave = 7.5
   523→        shockwaves.current.push({ r: 2.2, speed: 7.5, hit: false })
   524→      }
   525→    }
   526→    shockwaves.current = shockwaves.current.filter((w) => {
   527→      w.r += w.speed * dt
   528→      if (!w.hit) {
   529→        const d = Math.hypot(pos.current.x, pos.current.z)
   530→        if (Math.abs(d - w.r) < 0.85) {
   531→          w.hit = true
   532→          hurtPlayer(0.1)
   533→        }
   534→      }
   535→      return w.r <= PLATFORM_R + 4
   536→    })
   537→    // Rendu des ondes via refs (aucun re-render par frame)
   538→    for (let i = 0; i < 3; i++) {
   539→      const mesh = waveRefs.current[i]
   540→      if (!mesh) continue
   541→      const w = shockwaves.current[i]
   542→      if (w) {
   543→        mesh.visible = true
   544→        const s = w.r
   545→        mesh.scale.set(s, s, 1)
   546→      } else {
   547→        mesh.visible = false
   548→      }
   549→    }
   550→    bossCombat.shockwave = shockwaves.current[0]?.r ?? 0
   551→
   552→    // ── Attaque du joueur (fenêtre uniquement) ──
   553→    if (combatActions.playerAttack > 0) {
   554→      const hits = combatActions.playerAttack
   555→      combatActions.playerAttack = 0
   556→      if (bossCombat.window) {
   557→        for (let i = 0; i < hits; i++) hitBoss()
   558→      }
   559→    }
   560→
   561→    // ── Transitions de phase (seuils officiels 66 % / 33 %) ──
   562→    const hpPct = bossCombat.bossHp / bossCombat.bossMaxHp
   563→    const phaseDur = bossCombat.elapsed - bossCombat.phaseStart
   564→    if (phase === 1 && (hpPct <= 0.66 || bossCombat.bossHp <= 0) && phaseDur >= 8.2) {
   565→      bossCombat.phase = 2
   566→      bossCombat.phaseTimes.push(phaseDur)
   567→      bossCombat.phaseStart = bossCombat.elapsed
   568→      bossCombat.playerHp = Math.min(bossCombat.playerMaxHp, bossCombat.playerHp + bossCombat.playerMaxHp * 0.3)
   569→    } else if (phase === 2 && (hpPct <= 0.33 || bossCombat.bossHp <= 0) && phaseDur >= 8.2) {
   570→      bossCombat.phase = 3
   571→      bossCombat.phaseTimes.push(phaseDur)
   572→      bossCombat.phaseStart = bossCombat.elapsed
   573→      bossCombat.playerHp = Math.min(bossCombat.playerMaxHp, bossCombat.playerHp + bossCombat.playerMaxHp * 0.3)
   574→    }
   575→
   576→    // ── Transforms ──
   577→    if (playerG.current) {
   578→      playerG.current.position.copy(pos.current)
   579→      playerG.current.rotation.y = Math.atan2(-pos.current.x, -6 - pos.current.z)
   580→    }
   581→    // Caméra : cadre le joueur ET le boss (regard vers le haut de la forteresse)
   582→    const cd = cam.current.dist
   583→    const cx = pos.current.x - Math.sin(cam.current.yaw) * cd * Math.cos(cam.current.pitch)
   584→    const cz = pos.current.z - Math.cos(cam.current.yaw) * cd * Math.cos(cam.current.pitch)
   585→    const cy = 1.6 + Math.sin(cam.current.pitch) * cd
   586→    camera.position.lerp(new THREE.Vector3(cx, cy, cz), 0.2)
   587→    camera.lookAt(pos.current.x * 0.3, 5.2, -5.5)
   588→
   589→    // Synchronisation HUD (10 Hz via state externe)
   590→    T.force -= dt
   591→    if (T.force <= 0) {
   592→      T.force = 0.1
   593→      force((n) => n + 1)
   594→    }
   595→  })
   596→
   597→  const windowNow = bossCombat.window
   598→
   599→  return (
   600→    <group>
   601→      <Fortress />
   602→      <AetherionModel phase={bossCombat.phase} window={windowNow} />
   603→
   604→      {/* Étiquette du boss */}
   605→      <Html center distanceFactor={30} position={[0, 17.5, -9]} zIndexRange={[20, 0]}>
   606→        <div className="whitespace-nowrap rounded-sm px-3 py-1 text-center text-[15px] font-black tracking-[0.2em] text-[#d8f0ff]"
   607→          style={{ background: '#0c0a14cc', border: '1px solid #58a8f088', textShadow: '0 0 8px #58c8f0' }}>
   608→          AETHERION — LE ROI DU CIEL
   609→          <div className="text-[9px] font-semibold uppercase tracking-[0.3em] text-[#88c8f0]">Suprême · Boss Mondial</div>
   610→        </div>
   611→      </Html>
   612→
   613→      {/* Joueur */}
   614→      <group ref={playerG} position={[0, 0, 7]}>
   615→        {playerRace && playerApp && (
   616→          <AvatarModel race={playerRace} appearance={playerApp} equipment={playerEq} animated moveRef={moveRef} />
   617→        )}
   618→        <mesh rotation={[-Math.PI / 2, 0, 0]} position={[0, 0.06, 0]}>
   619→          <ringGeometry args={[0.55, 0.72, 24]} />
   620→          <meshBasicMaterial color="#58c878" transparent opacity={0.8} />
   621→        </mesh>
   622→      </group>
   623→
   624→      {/* Télégraphes d'éclairs */}
   625→      {strikesView.map((s) => {
   626→        const remaining = s.t - bossCombat.elapsed
   627→        const imminent = remaining < 0.35
   628→        return (
   629→          <group key={s.id}>
   630→            <mesh rotation={[-Math.PI / 2, 0, 0]} position={[s.x, 0.08, s.z]}>
   631→              <ringGeometry args={[s.r - 0.5, s.r, 26]} />
   632→              <meshBasicMaterial color={imminent ? '#ffffff' : '#58a8f0'} transparent opacity={imminent ? 0.95 : 0.7} />
   633→            </mesh>
   634→            {imminent && (
   635→              <mesh position={[s.x, 6, s.z]}>
   636→                <cylinderGeometry args={[0.32, 0.55, 12, 8, 1, true]} />
   637→                <meshBasicMaterial color="#bfe4ff" transparent opacity={0.85} side={THREE.DoubleSide} />
   638→              </mesh>
   639→            )}
   640→          </group>
   641→        )
   642→      })}
   643→
   644→      {/* Ondes de choc (phase 2) — pool de 3 meshes pilotés par refs */}
   645→      {[0, 1, 2].map((i) => (
   646→        <mesh
   647→          key={`wave${i}`}
   648→          ref={(el) => {
   649→            waveRefs.current[i] = el
   650→          }}
   651→          rotation={[-Math.PI / 2, 0, 0]}
   652→          position={[0, 0.14, 0]}
   653→          visible={false}
   654→        >
   655→          <ringGeometry args={[0.9, 1, 48]} />
   656→          <meshBasicMaterial color="#88b8f0" transparent opacity={0.55} />
   657→        </mesh>
   658→      ))}
   659→    </group>
   660→  )
   661→}

   663→function BossScene({ encounter, onEnd }: { encounter: BossEncounter; onEnd: (r: 'victory' | 'defeat') => void }) {
   664→  const phase = bossCombat.phase
   665→  const spiritual = phase >= 3
   666→  return (
   667→    <>
   668→      <color attach="background" args={[spiritual ? '#1c2440' : '#5c94d8']} />
   669→      <fog attach="fog" args={[spiritual ? '#243055' : '#a8cdf0', 40, 150]} />
   670→      <ambientLight intensity={spiritual ? 0.65 : 1.1} color={spiritual ? '#8898d8' : '#ffffff'} />
   671→      <directionalLight position={[20, 34, 16]} intensity={1.6} color="#fff4e0" castShadow={false} />
   672→      <directionalLight position={[-14, 10, -18]} intensity={0.8} color="#88c8f0" />
   673→      <SkyGradient spiritual={spiritual} />
   674→      <BossFight encounter={encounter} onEnd={onEnd} />
   675→    </>
   676→  )
   677→}

   679→function SkyGradient({ spiritual }: { spiritual: boolean }) {
   680→  const texture = useMemo(() => {
   681→    const c = document.createElement('canvas')
   682→    c.width = 2
   683→    c.height = 256
   684→    const ctx = c.getContext('2d')!
   685→    const g = ctx.createLinearGradient(0, 0, 0, 256)
   686→    g.addColorStop(0, spiritual ? '#0c1228' : '#3c78c8')
   687→    g.addColorStop(0.6, spiritual ? '#28345c' : '#88bce8')
   688→    g.addColorStop(1, spiritual ? '#384470' : '#e8f2fc')
   689→    ctx.fillStyle = g
   690→    ctx.fillRect(0, 0, 2, 256)
   691→    const tex = new THREE.CanvasTexture(c)
   692→    tex.colorSpace = THREE.SRGBColorSpace
   693→    return tex
   694→  }, [spiritual])
   695→  return (
   696→    <mesh>
   697→      <sphereGeometry args={[170, 24, 16]} />
   698→      <meshBasicMaterial map={texture} side={THREE.BackSide} depthWrite={false} fog={false} />
   699→    </mesh>
   700→  )
   701→}
